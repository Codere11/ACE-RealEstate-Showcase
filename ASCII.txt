    DATA MODEL (backend)

  bookings table:
    id              SERIAL PK
    organization_id FK → organizations
    lead_id         FK → leads (nullable, for AI-booked)
    customer_name   VARCHAR(160)
    customer_phone  VARCHAR(30)
    customer_email  VARCHAR(160)
    service_id      VARCHAR(40)   ('nega-obraza'|'maska-obraza'|'ciscenje-obraza')
    booking_date    DATE
    booking_time    TIME
    duration_min    INTEGER
    price_eur       INTEGER
    status          VARCHAR(20)   ('confirmed'|'in_progress'|'completed'|'no_show'|'cancelled')
    notes           TEXT
    created_at      TIMESTAMPTZ
    updated_at      TIMESTAMPTZ
